# seller/ — deterministic AAMP seller MCP server (runtime deps)
#
# The seller REUSES the IAB seller-agent SDK's models + engines (MediaKitService,
# PricingRulesEngine) VERBATIM, but runs NO agents — so it does NOT need the SDK's
# heavy `crewai` stack. We install the SDK with `--no-deps` and pin these slim deps.
#
# Dev / CI install (crewai-free):
#   pip install -r requirements.txt
#   pip install --no-deps "ad_seller_system @ git+https://github.com/IABTechLab/seller-agent.git@0b463ba"
#
# (The pinned SDK commit is 0b463ba. See seller/README.md for the deploy story.)

# --- our server + data access ---

fastapi>=0.115.0

psycopg[binary,pool]>=3.1  # relational reads (aamp.pricing_rules) + a reused connection pool
